## 📋 开发前准备

### 1. 了解现有项目结构
```
applications/
├── view/
│   ├── system/          # PC端视图
│   │   ├── ticket.py    # 工单管理
│   │   ├── passport.py  # 登录认证
│   │   └── ...
│   └── api/             # API接口
├── models/              # 数据模型
├── extensions/          # 扩展模块
└── config.py           # 配置文件

templates/
├── system/             # PC端模板
│   ├── ticket/         # 工单模板
│   │   ├── main.html   # 工单列表
│   │   ├── add.html    # 新增工单
│   │   ├── edit.html   # 编辑工单
│   │   └── view.html   # 查看工单
│   ├── passport/       # 登录模板
│   └── ...

static/
├── css/                # 样式文件
├── js/                 # JavaScript文件
└── system/             # 系统资源
```

### 2. 移动端技术选型
- **UI框架**: Layui Mobile (已包含在项目中)
- **布局方式**: 响应式布局 + 移动端专用模板
- **适配策略**: 独立移动端路由，复用后端逻辑
- **功能范围**: 仅工单管理（登录、列表、新增、查看、编辑、删除）

### 步骤 1.4: 创建移动端公共JS

**创建文件**: `static/js/mobile/common.js`

**需要写入的内容**:
```javascript
/**
 * 移动端公共JS
 */

// 配置Layui
layui.config({
    version: '1.0.0'
});

// 移动端公共方法
var MobileCommon = {
    /**
     * 显示加载中
     */
    showLoading: function(message) {
        layui.layer.open({
            type: 2,
            content: message || '加载中...',
            shadeClose: false
        });
    },
    
    /**
     * 隐藏加载中
     */
    hideLoading: function() {
        layui.layer.closeAll('loading');
    },
    
    /**
     * 显示提示消息
     */
    showToast: function(message, icon) {
        layui.layer.open({
            content: message,
            skin: 'msg',
            time: 2
        });
    },
    
    /**
     * 显示确认对话框
     */
    showConfirm: function(message, callback) {
        layui.layer.open({
            content: message,
            btn: ['确定', '取消'],
            yes: function(index) {
                layui.layer.close(index);
                if (callback) callback(true);
            },
            no: function(index) {
                layui.layer.close(index);
                if (callback) callback(false);
            }
        });
    },
    
    /**
     * 格式化日期时间
     */
    formatDate: function(dateStr) {
        if (!dateStr) return '';
        var date = new Date(dateStr);
        var month = (date.getMonth() + 1).toString().padStart(2, '0');
        var day = date.getDate().toString().padStart(2, '0');
        var hours = date.getHours().toString().padStart(2, '0');
        var minutes = date.getMinutes().toString().padStart(2, '0');
        return month + '-' + day + ' ' + hours + ':' + minutes;
    },
    
    /**
     * 获取优先级标签
     */
    getPriorityTag: function(priority) {
        var map = {
            'P1': { text: 'P1-紧急', class: 'danger' },
            'P2': { text: 'P2-高', class: 'warning' },
            'P3': { text: 'P3-中', class: 'primary' },
            'P4': { text: 'P4-低', class: 'success' }
        };
        return map[priority] || { text: priority, class: '' };
    },
    
    /**
     * 获取状态标签
     */
    getStatusTag: function(status) {
        var map = {
            '待处理': { text: '待处理', class: 'warning' },
            '处理中': { text: '处理中', class: 'primary' },
            '已完成': { text: '已完成', class: 'success' },
            '已关闭': { text: '已关闭', class: '' }
        };
        return map[status] || { text: status, class: '' };
    },
    
    /**
     * AJAX请求封装
     */
    request: function(options) {
        var self = this;
        
        // 显示加载中
        if (options.loading !== false) {
            self.showLoading();
        }
        
        layui.$.ajax({
            url: options.url,
            type: options.type || 'GET',
            data: options.data || {},
            dataType: 'json',
            success: function(res) {
                self.hideLoading();
                if (res.code === 0 || res.success) {
                    if (options.success) options.success(res);
                } else {
                    self.showToast(res.msg || '操作失败');
                    if (options.error) options.error(res);
                }
            },
            error: function() {
                self.hideLoading();
                self.showToast('网络错误，请稍后重试');
                if (options.error) options.error();
            }
        });
    }
};

// 全局变量
window.MobileCommon = MobileCommon;
```

```python
from flask import Blueprint, render_template, request, redirect, url_for, session
from flask_login import login_user
from applications.models import User
from applications.extensions import db
from applications.common.admin import get_captcha
from applications.common.utils.http import fail_api, success_api

bp = Blueprint('mobile_passport', __name__)

@bp.route('/login', methods=['GET', 'POST'])
def login():
    """移动端登录页面"""
    if request.method == 'GET':
        # 如果已登录，跳转到首页
        if session.get('_user_id'):
            return redirect(url_for('mobile.ticket_list'))
        return render_template('mobile/passport/login.html')
    
    # POST 请求处理登录
    req = request.form
    username = req.get('username')
    password = req.get('password')
    code = req.get('captcha', '').lower()
    
    # 验证参数
    if not username or not password or not code:
        return fail_api(msg="请填写完整信息")
    
    # 验证验证码
    s_code = session.get("code", None)
    if not s_code or code != s_code.lower():
        return fail_api(msg="验证码错误")
    
    # 查询用户
    user = User.query.filter_by(username=username).first()
    if not user:
        return fail_api(msg="用户名或密码错误")
    
    # 验证密码
    if not user.validate_password(password):
        return fail_api(msg="用户名或密码错误")
    
    # 检查用户状态
    if not user.enable:
        return fail_api(msg="账号已被禁用")
    
    # 登录成功
    login_user(user)
    session.permanent = True
    
    return success_api(msg="登录成功")

@bp.route('/logout')
def logout():
    """退出登录"""
    from flask_login import logout_user
    logout_user()
    return redirect(url_for('mobile_passport.login'))
```

### 步骤 3.1: 创建移动端工单列表视图

**创建文件**: `applications/view/mobile/ticket.py`

**需要写入的内容**:
```python
from flask import Blueprint, render_template, request, jsonify
from flask_login import login_required, current_user
from applications.models import Ticket, User
from applications.extensions import db
from applications.common.utils.http import fail_api, success_api
from sqlalchemy import or_

bp = Blueprint('mobile_ticket', __name__)

@bp.route('/tickets')
@login_required
def ticket_list():
    """移动端工单列表页面"""
    return render_template('mobile/ticket/list.html')

@bp.route('/api/tickets')
@login_required
def api_ticket_list():
    """移动端工单列表API"""
    # 获取分页参数
    page = request.args.get('page', 1, type=int)
    limit = request.args.get('limit', 10, type=int)
    
    # 获取筛选参数
    keyword = request.args.get('keyword', '')
    priority = request.args.get('priority', '')
    status = request.args.get('status', '')
    
    # 构建查询
    query = Ticket.query
    
    # 关键词搜索
    if keyword:
        query = query.filter(
            or_(
                Ticket.title.contains(keyword),
                Ticket.id.contains(keyword),
                Ticket.customer_name.contains(keyword)
            )
        )
    
    # 优先级筛选
    if priority:
        query = query.filter(Ticket.priority == priority)
    
    # 状态筛选
    if status:
        query = query.filter(Ticket.status == status)
    
    # 排序和分页
    query = query.order_by(Ticket.create_time.desc())
    pagination = query.paginate(page=page, per_page=limit, error_out=False)
    
    # 格式化数据
    items = []
    for ticket in pagination.items:
        items.append({
            'id': ticket.id,
            'title': ticket.title,
            'priority': ticket.priority,
            'status': ticket.status,
            'customer_name': ticket.customer_name,
            'handler_name': ticket.handler_name,
            'create_time': ticket.create_time.strftime('%m-%d %H:%M') if ticket.create_time else ''
        })
    
    return jsonify({
        'code': 0,
        'data': items,
        'total': pagination.total,
        'has_more': pagination.has_next
    })

@bp.route('/ticket/<int:ticket_id>')
@login_required
def ticket_view(ticket_id):
```

### 步骤 3.3: 创建移动端工单列表JS

**创建文件**: `static/js/mobile/ticket/ticket_list.js`

**需要写入的内容**:
```javascript
/**
 * 移动端工单列表JS
 */

layui.use(['layer', 'jquery'], function(){
    var layer = layui.layer;
    var $ = layui.jquery;
    
    // 当前页码
    var currentPage = 1;
    var isLoading = false;
    var hasMore = true;
    
    // 筛选条件
    var filters = {
        keyword: '',
        priority: '',
        status: ''
    };
    
    /**
     * 加载工单列表
     */
    function loadTickets(page, append) {
        if (isLoading || (!hasMore && append)) return;
        
        isLoading = true;
        
        if (!append) {
            MobileCommon.showLoading('加载中...');
        }
        
        $.ajax({
            url: '/m/api/tickets',
            type: 'GET',
            data: {
                page: page,
                limit: 10,
                keyword: filters.keyword,
                priority: filters.priority,
                status: filters.status
            },
            dataType: 'json',
            success: function(res) {
                MobileCommon.hideLoading();
                isLoading = false;
                
                if (res.code === 0) {
                    hasMore = res.has_more;
                    renderTickets(res.data, append);
                    
                    if (res.data.length === 0 && !append) {
                        $('#emptyState').show();
                        $('#ticketList').hide();
                    } else {
                        $('#emptyState').hide();
                        $('#ticketList').show();
                    }
                    
                    // 显示/隐藏加载更多
                    if (hasMore) {
                        $('#loadMore').show().text('上拉加载更多');
                    } else {
                        $('#loadMore').show().text('没有更多了');
                    }
                }
            },
            error: function() {
                MobileCommon.hideLoading();
                isLoading = false;
                MobileCommon.showToast('加载失败');
            }
        });
    }
    
    /**
     * 渲染工单列表
     */
    function renderTickets(tickets, append) {
        var html = '';
        
        tickets.forEach(function(ticket) {
            var priorityClass = 'priority-' + ticket.priority;
            var statusClass = 'status-' + ticket.status;
            
            html += '<div class="ticket-card" data-id="' + ticket.id + '">' +
                '<div class="ticket-header">' +
                    '<div class="ticket-title">' + ticket.title + '</div>' +
                    '<span class="ticket-priority ' + priorityClass + '">' + ticket.priority + '</span>' +
                '</div>' +
                '<div class="ticket-info">' +
                    '<span>' + ticket.customer_name + ' | ' + ticket.create_time + '</span>' +
                    '<span class="ticket-status ' + statusClass + '">' + ticket.status + '</span>' +
                '</div>' +
            '</div>';
        });
        
        if (append) {
            $('#ticketList').append(html);
        } else {
            $('#ticketList').html(html);
        }
    }
    
    /**
     * 绑定事件
     */
    function bindEvents() {
        // 搜索输入
        var searchTimer;
        $('#searchInput').on('input', function() {
            clearTimeout(searchTimer);
            filters.keyword = $(this).val();
            searchTimer = setTimeout(function() {
                currentPage = 1;
                loadTickets(currentPage, false);
            }, 500);
        });
        
        // 筛选标签点击
        $('.filter-item').on('click', function() {
            $(this).addClass('active').siblings().removeClass('active');
            
            var filter = $(this).data('filter');
            
            // 重置筛选条件
            filters.priority = '';
            filters.status = '';
            
            if (filter === 'all') {
                // 全部
            } else if (filter.startsWith('P')) {
                filters.priority = filter;
            } else {
                filters.status = filter;
            }
            
            currentPage = 1;
            loadTickets(currentPage, false);
        });
        
        // 工单卡片点击
        $('#ticketList').on('click', '.ticket-card', function() {
            var id = $(this).data('id');
            window.location.href = '/m/ticket/' + id;
        });
        
        // 上拉加载更多（使用滚动事件模拟）
        $(window).on('scroll', function() {
            if ($(window).scrollTop() + $(window).height() >= $(document).height() - 50) {
                if (hasMore && !isLoading) {
                    currentPage++;
                    loadTickets(currentPage, true);
                }
            }
        });
    }
    
    // 初始化
    loadTickets(1, false);
    bindEvents();
});
```

```html
function deleteTicket(id) {
    MobileCommon.showConfirm('确定要删除这个工单吗？', function(confirmed) {
        if (confirmed) {
            MobileCommon.showLoading('删除中...');
            
            layui.$.ajax({
                url: '/m/api/ticket/' + id + '/delete',
                type: 'POST',
                dataType: 'json',
                success: function(res) {
                    MobileCommon.hideLoading();
                    if (res.code === 0 || res.success) {
                        layui.layer.msg('删除成功', {icon: 1});
                        setTimeout(function() {
                            window.location.href = "{{ url_for('mobile.ticket_list') }}";
                        }, 1000);
                    } else {
                        layui.layer.msg(res.msg || '删除失败', {icon: 2});
                    }
                },
                error: function() {
                    MobileCommon.hideLoading();
                    layui.layer.msg('网络错误', {icon: 2});
                }
            });
        }
    });
}
</script>
```
